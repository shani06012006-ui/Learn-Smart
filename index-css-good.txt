/* ===== Landing: character + floating objects ===== */

/* Character entrance and idle motion.
   Two layers, so transition (entrance) and animation (idle float) never
   fight over `transform`:
     - .landing-character       — outer wrapper, handles entrance transition
     - .landing-character-art   — inner wrapper, handles idle float
*/
.landing-character {
  opacity: 0;
  transform: translateY(24px) scale(0.96);
  transition: opacity 900ms cubic-bezier(0.2, 0.6, 0.2, 1),
              transform 900ms cubic-bezier(0.2, 0.6, 0.2, 1);
}
.landing-character.is-revealed {
  opacity: 1;
  transform: translateY(0) scale(1);
}
.landing-character.is-revealed .landing-character-art,
.landing-character.is-revealed .landing-character-placeholder {
  animation: character-idle 6s ease-in-out infinite;
}

@keyframes character-idle {
  0%, 100% { transform: translateY(0); }
  50%      { transform: translateY(-8px); }
}

/* Poses â€” subtle changes to suggest a gesture. These are designed to work
   with the placeholder silhouette AND with any real illustration that
   uses the same bounding box, because they only rotate/translate the
   wrapper. */
.landing-character.character--pointing .landing-character-placeholder {
  transform: rotate(-3deg) translateX(-2px);
}
.landing-character.character--studying .landing-character-placeholder {
  transform: rotate(2deg) translateY(4px);
}
.landing-character.character--celebrating .landing-character-placeholder {
  animation: character-celebrate 1.2s ease-in-out;
}
.landing-character.character--joining .landing-character-placeholder {
  transform: translateX(4px);
}

@keyframes character-celebrate {
  0%   { transform: translateY(0) scale(1); }
  30%  { transform: translateY(-14px) scale(1.03); }
  60%  { transform: translateY(-4px) scale(1.01); }
  100% { transform: translateY(0) scale(1); }
}

/* Soft glow behind the character */
.landing-character-glow {

    radial-gradient(closest-side, rgb(165 180 252 / 0.12), transparent 60%);

/* Floating learning objects â€” fade/slide in, then idle */
